\begin{frame}{Capacitancia}{Definición}
    \begin{boxTeoria}{Capacitancia ($C$)}
        Mide la capacidad de almacenar carga por unidad de voltaje. Depende solo de la geometría y del medio.
        \begin{equation*}
            C = \frac{Q}{|\Delta V|} \quad [\unit{\F}]
        \end{equation*}
    \end{boxTeoria}
\end{frame}

\begin{frame}{Obtención de Capacitancia}
    \begin{boxTeoria}{Algoritmo de Cálculo (4 Pasos)}
        \begin{enumerate}
            \item Asumir carga $+Q$ y $-Q$ en los conductores.
            \item Hallar $\vec{E}$ con Ley de Gauss: $\oint \vec{E} \cdot d\vec{A} = \frac{Q}{\epsilon_0}$
            \item Hallar $\Delta V$ integrando $\vec{E}$: $\Delta V = -\int \vec{E} \cdot d\vec{l}$
            \item Calcular $C = Q / |\Delta V|$ (La $Q$ siempre se cancela).
        \end{enumerate}
    \end{boxTeoria}
\end{frame}

\begin{frame}{Energía y Circuitos}
    \begin{boxTeoria}{4. Energía ($U$) y Circuitos}
        \begin{itemize}
            \item \textbf{Energía almacenada:} $U = \frac{1}{2} C (\Delta V)^2 = \frac{Q^2}{2C}$
            \item \textbf{Condensadores en Serie:} $\frac{1}{C_{eq}} = \frac{1}{C_1} + \frac{1}{C_2} + \dots$ (Misma $Q$)
            \item \textbf{Condensadores en Paralelo:} $C_{eq} = C_1 + C_2 + \dots$ (Mismo $V$)
        \end{itemize}
    \end{boxTeoria}

\end{frame}

\begin{frame}{Problema 2}

    \begin{enumerate}[label=\alph*)]
        \item La intensidad de campo eléctrico en la región vacía entre las armaduras.
        \item La intensidad de campo eléctrico en el dieléctrico.
    \end{enumerate}
