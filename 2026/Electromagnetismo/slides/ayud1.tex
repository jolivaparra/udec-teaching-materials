\begin{frame}{Operaciones Vectoriales}{Repaso}
    Dado dos vectores: $\vect{r}_1$, $\vect{r}_2$ se tiene lo siguiente.
    \begin{boxTeoria}{Vector Posición}
        El vector $\vect{r}_{12}$, corresponde al vector que va desde $\vect{r}_1$ hasta $\vect{r}_2$ está dado por:
        \begin{equation*}
            \vect{r}_{12} = \vect{r}_2 - \vect{r}_1
        \end{equation*}
    \end{boxTeoria}
\end{frame}

\begin{frame}{Operaciones Vectoriales}{Repaso}
    \vspace{-15pt}
    \begin{boxTeoria}{Norma de un Vector}
        Dado $\vect{r} = r_x\hat{x} + r_y\hat{y}$ la norma del vector $\vect{r}$ está dada por:
        \begin{equation*}
            |\vect{r}| = r = \sqrt{r_x^2 + r_y^2}
        \end{equation*}
    \end{boxTeoria}

    \begin{boxTeoria}{Vector Unitario}
        Un vector unitario es aquel que tiene norma 1. Para normalizar un vector basta dividirlo por su propia norma.
        \begin{equation*}
            \hat{r} = \frac{\vect{r}}{|\vect{r}|}
        \end{equation*}
    \end{boxTeoria}
\end{frame}

\begin{frame}{Definiciones}

    \begin{boxTeoria}{Carga Eléctrica}
        Es una propiedad intrínseca de la materia, al igual que la masa, que determina la capacidad de las partículas para participar en interacciones electromagnéticas.
    \end{boxTeoria}

    \begin{boxTeoria}{Carga Puntual}
        Es una idealización física donde se asume que toda la carga eléctrica de un cuerpo está concentrada en un único punto geométrico.
        \begin{itemize}
            \item Se suele utilizar cuando la distancia entre los cuerpos cargados es mucho mayor que el tamaño de los cuerpos mismos.
        \end{itemize}
    \end{boxTeoria}

\end{frame}

\begin{frame}{Ley de Coulomb}
    Establece que la fuerza entre dos cargas puntuales en reposo es directamente proporcional al producto de sus cargas e inversamente proporcional al cuadrado de la distancia que las separa.
    \begin{boxTeoria}{Expresión Vectorial}
        La fuerza ejercida por la carga $q_1$ ubicada en $\vect{r}_1$ sobre $q_2$ ubicada en $\vect{r_2}$ se define como:
        \begin{equation*}
            \vec{F}_{12} = \frac{1}{4\pi\varepsilon_0} \frac{\dotprod{q_1}{q_2}}{|\vect{r}_{12}|^2} \;\hat{r}_{12} \;[\unit{\N}]
        \end{equation*}
    \end{boxTeoria}

    \begin{itemize}
        \item \textbf{Dirección:} Actúa a lo largo de la línea recta que une ambas cargas.
        \item \textbf{Sentido:} Es atractiva si las cargas son de signo opuesto ($q_1 q_2 < 0$) y repulsiva si son del mismo signo ($q_1 q_2 > 0$).
        \item \textbf{Constante:} $k_e = \frac{1}{4\pi\varepsilon_0}$
    \end{itemize}
\end{frame}

\begin{frame}{Utilidades}
    \vspace{-15pt}
    \begin{boxTeoria}{Principio de Superposición}
        La fuerza neta que actúa sobre una carga puntual es la suma vectorial de las fuerzas ejercidas por cada una de las demás cargas del sistema.
        \begin{equation*}
            \vect{F}_{R} = \vect{F}_{1} + \vect{F}_{2} + \dots + \vect{F}_{n}
        \end{equation*}
    \end{boxTeoria}

    \begin{boxNota}{Valores de Algunas Constantes}
        \begin{itemize}
            \item \textbf{Magnitud de la Carga del Electrón ($e$):} \qty{1.602e-19}{\C}
            \item \textbf{Permitividad del Vacío ($\epsilon_0$):} \qty{8.854e-12}{\C\squared\per\N\per\meter\squared}
            \item \textbf{Masa del electrón ($m_e$):} \qty{9.109e-31}{\kg}
        \end{itemize}
    \end{boxNota}

\end{frame}

\begin{frame}{Formas de Distribuciones de Carga}

    \begin{boxTeoria}{Distribución Lineal de Carga}
        Es la razón entre la cantidad de carga ($\Delta q$) y la longitud ($\Delta l$) de un cuerpo.
        Define cuanta carga existe por cada unidad de longitud ($dL$).
        \begin{equation*}
            \rho_l = \lambda = \lim_{\Delta l\to 0}\frac{\Delta q}{\Delta l} = \frac{dq}{dl} \;[\unit{\C\per\m}]
        \end{equation*}
    \end{boxTeoria}
\end{frame}

\begin{frame}{Formas de Distribuciones de Carga}

    \begin{columns}
        \begin{column}{0.49\textwidth}
            \begin{boxTeoria}{Distribución de Carga Superficial}
                Es la razón entre la cantidad de carga ($\Delta q$) y la superficie ($\Delta S$) de un cuerpo.
                Define cuanta carga existe por cada unidad de superficie ($dS$).
                \begin{equation*}
                    \rho_s = \sigma = \lim_{\Delta S\to 0}\frac{\Delta q}{\Delta S} = \frac{dq}{dS} \;[\unit{\C\per\m\squared}]
                \end{equation*}
            \end{boxTeoria}
        \end{column}
        \hfill
        \begin{column}{0.49\textwidth}
            \begin{boxTeoria}{Distribución de Carga Volumétrica}
                Es la razón entre la cantidad de carga ($\Delta q$) y el volumen ($\Delta V$) de un cuerpo.
                Define cuanta carga existe por cada unidad de volumen ($dV$).
                \begin{equation*}
                    \rho_s = \rho = \lim_{\Delta V\to 0}\frac{\Delta q}{\Delta V} = \frac{dq}{dV} \;[\unit{\C\per\m\cubed}]
                \end{equation*}
            \end{boxTeoria}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{Campo Eléctrico}
    \begin{boxTeoria}{Definición}
        Si $\vec{F}$ es la fuerza ejercida sobre una carga de prueba $q_0$ ubicada en la posición $\vec{r}$, el campo eléctrico en dicho punto se define como el límite:
        \begin{equation*}
            \vec{E}(\vec{r}) = \lim_{q_0 \to 0} \frac{\vec{F}}{q_0}
        \end{equation*}
    \end{boxTeoria}
\end{frame}


\begin{frame}{Integración por Elementos Diferenciales}
    Las fórmulas básicas (como la de Newton o Coulomb) son puntuales (trabajan punto a punto). Pero en el mundo real, los objetos tienen forma y tamaño. Para estudiar objetos reales, los tratamos como una colección infinita de puntos.

    \begin{boxTeoria}{Proceso}
        Consiste en 2 pasos principales (no necesariamente fáciles)
        \begin{enumerate}
            \item \textbf{Dividir:} Cortar el objeto en trozos infinitesimales llamados \textbf{elementos diferenciales} ($dx$, $dA$, $dV$).
            \item \textbf{Aproximar:} Cada trozo es tan pequeño que podemos tratarlo como un ``punto'' con una porción de propiedad total ($dq$ para carga, $dm$ para masa).
        \end{enumerate}

    \end{boxTeoria}
\end{frame}

\begin{frame}{Integración por Elementos Diferenciales}
    Para pasar de lo pequeño a lo grande, sumamos los infinitos puntos involucrados a lo largo de toda su extensión mediante la \textbf{Integral}.

    \begin{equation*}
        \text{Cantidad Total} = \int_\text{objeto}d(\text{Cantidad})
    \end{equation*}

\end{frame}
