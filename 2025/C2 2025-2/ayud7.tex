\begin{frame}{Superficie de sólidos de revolución}
\framesubtitle{Definición}
    \begin{definicion}
        La expresión general para calcular el área de la superficie de un sólido de revolución debido a una curva $y = f(x)$ son
        \begin{align*}
            S = \int_a^b 2\pi r(x)\sqrt{1+(f'(x))^2}\,dx\quad
        \end{align*}
    \end{definicion}
    donde $r(x)$ corresponde a los segmentos entre la curva de interés y el eje de rotación.
\end{frame}

\begin{frame}{Parametrizaciones Típicas}
    \begin{itemize}
        \item Segmento de recta entre $A$ y $B$:\\[-15pt]
        \[ \vec{r}(t) = Bt + (1-t)A, \quad t\in[0,1]\]
        \item Circunferencia de radio R centrada en $C=(x_0, y_0)$:\\[-15pt]
        \[\vec{r}(t) = (x_0+R\cos(t),y_0+R\sin(t)), \quad t\in[0,2\pi]\]
        \item Elipse de radio mayor $a$ en eje $x$, y radio menor $b$ en eje y, con centro $C=(x_0, y_0)$.\\[-15pt]
        \[\vec{r}(t) = (x_0+a\cos(t),y_0+b\sin(t)), \quad t\in[0,2\pi]\]
    \end{itemize}
\end{frame}

\begin{frame}{Longitud de Arco en Curvas Paramétricas}
\framesubtitle{Definición}
    \begin{definicion}
        Sea una curva paramétrica $C: \vec{r}=(x(t),y(t)), t\in[a,b]$, la longitud de arco de la gráfica de $C$ entre los puntos $\vec{r}(a)$ y $\vec{r}(b)$ se da por.
        \begin{equation*}
            L = \int_a^b \sqrt{[x'(t)]^2+[y'(t)]^2}\,dx
        \end{equation*}
    \end{definicion}
\end{frame}
